#!/bin/sh
# router_nvram_show.sh
#
# ASUS-Merlin (BusyBox) friendly export:
#   1) output/router_nvram.txt   (labelled nvram dump)
#   2) output/router_dhcp.csv    (merged view of reservations + custom names + active leases)
#   3) output/router_lists_raw.txt (raw nvram lists for dhcp/custom/leases for debugging)

# Run: sh router_nvram_show.sh

# Notes on formats (varies by firmware):
# - Lists are commonly: entry<entry<entry with fields separated by >
# - We'll parse using awk FS='[<>]' and build a union keyed by MAC.
#
# Output columns:
# mac, ip_reserved, ip_current, name_reserved, name_custom, name_lease, name_final, sources
#
# "sources" is a pipe-separated string of where the row came from: reserved|custom|lease

escape_csv() {
  # Input: arbitrary string on stdin; Output: CSV-safe quoted field
  # - doubles embedded quotes
  # - wraps in quotes
  awk '{
    gsub(/"/, "\"\"");
    printf "\"%s\"", $0
  }'
}

# Create temp files (BusyBox-safe)
TMP_DIR="${OUT_DIR}/.tmp_router_nvram"
rm -rf "$TMP_DIR"
mkdir -p "$TMP_DIR"

STATIC_TSV="$TMP_DIR/static.tsv"   # mac \t ip \t name
CUSTOM_TSV="$TMP_DIR/custom.tsv"   # mac \t name
LEASE_TSV="$TMP_DIR/lease.tsv"     # mac \t ip \t name

# Parse dhcp_staticlist -> TSV (best-effort: mac, ip, name)
printf '%s' "$dhcp_staticlist" | awk -v FS='[<>]' '
BEGIN { OFS="\t" }
{
  # Split entries separated by "<" are already seen as records when piped; but many firmwares store whole string.
}
' >/dev/null 2>&1

# Robust parse: replace "<" with newlines first, then split by ">"
printf '%s' "$dhcp_staticlist" | tr '<' '\n' | awk -F'>' '
BEGIN { OFS="\t" }
NF >= 2 {
  mac=$1; ip=$2; name="";
  if (NF >= 3) name=$3;
  # basic sanity: MAC contains :
  if (mac ~ /:/) print mac, ip, name;
}
' > "$STATIC_TSV"

# Parse custom_clientlist -> TSV (best-effort: mac, name)
printf '%s' "$custom_clientlist" | tr '<' '\n' | awk -F'>' '
BEGIN { OFS="\t" }
NF >= 2 {
  mac=$1; name=$2;
  if (mac ~ /:/) print mac, name;
}
' > "$CUSTOM_TSV"

# Parse dhcp_leases -> TSV (best-effort: mac, ip, name)
# Many builds use: <expire>><mac>><ip>><hostname>>...
# Others use: mac>ip>hostname>...
# We'll try both patterns.
printf '%s' "$dhcp_leases" | tr '<' '\n' | awk -F'>' '
BEGIN { OFS="\t" }
{
  # Remove empty fields caused by leading/trailing separators
  # Build an array of non-empty fields
  n=0;
  for (i=1; i<=NF; i++) {
    if ($i != "") { n++; f[n]=$i; }
  }
  if (n < 2) next;

  mac=""; ip=""; name="";

  # Pattern A: expire, mac, ip, name
  if (f[2] ~ /:/ && f[3] ~ /^[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+$/) {
    mac=f[2]; ip=f[3];
    if (n >= 4) name=f[4];
  }
  # Pattern B: mac, ip, name
  else if (f[1] ~ /:/ && f[2] ~ /^[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+$/) {
    mac=f[1]; ip=f[2];
    if (n >= 3) name=f[3];
  }

  if (mac ~ /:/ && ip != "") print mac, ip, name;
}
' > "$LEASE_TSV"

# Merge union by MAC
# Priority:
# - IP reserved preferred for "ip_reserved"
# - Current IP from lease (if present) for "ip_current"
# - Name final: reserved name > custom name > lease name
#
# Build a single awk merge from the three TSV files.
{
  echo "mac,ip_reserved,ip_current,name_reserved,name_custom,name_lease,name_final,sources"
  awk -F'\t' '
  BEGIN { OFS="," }

  function q(s,   t) {
    # CSV-quote
    gsub(/"/, "\"\"", s);
    return "\"" s "\"";
  }

  FNR==NR {
    # static.tsv: mac ip name
    mac=$1; ipr=$2; nrn=$3;
    ip_res[mac]=ipr;
    name_res[mac]=nrn;
    src[mac]="reserved";
    next
  }
  FILENAME ~ /custom\.tsv$/ {
    mac=$1; ncn=$2;
    name_cus[mac]=ncn;
    if (src[mac] == "") src[mac]="custom";
    else if (index("|" src[mac] "|", "|custom|") == 0) src[mac]=src[mac] "|custom";
    next
  }
  FILENAME ~ /lease\.tsv$/ {
    mac=$1; ipc=$2; nln=$3;
    ip_cur[mac]=ipc;
    name_lease[mac]=nln;
    if (src[mac] == "") src[mac]="lease";
    else if (index("|" src[mac] "|", "|lease|") == 0) src[mac]=src[mac] "|lease";
    next
  }
  END {
    # union keys: iterate over all arrays by scanning src[]
    for (mac in src) {
      ipr = (mac in ip_res) ? ip_res[mac] : "";
      ipc = (mac in ip_cur) ? ip_cur[mac] : "";
      nrn = (mac in name_res) ? name_res[mac] : "";
      ncn = (mac in name_cus) ? name_cus[mac] : "";
      nln = (mac in name_lease) ? name_lease[mac] : "";

      nf = "";
      if (nrn != "") nf = nrn;
      else if (ncn != "") nf = ncn;
      else if (nln != "") nf = nln;

      s = src[mac];

      print q(mac), q(ipr), q(ipc), q(nrn), q(ncn), q(nln), q(nf), q(s);
    }
  }
  ' "$STATIC_TSV" "$CUSTOM_TSV" "$LEASE_TSV"
} > "$DHCP_CSV"

# Cleanup
rm -rf "$TMP_DIR"

echo "Wrote:"
echo "  $NVRAM_TXT"
echo "  $DHCP_CSV"
echo "  $RAW_TXT"

# 03 – DHCP Reservations

## Source of truth

DHCP reservations are stored on the ASUS router (Merlin) in NVRAM (`dhcp_staticlist`) and exported via `router_nvram_show.sh`.

## DHCP pool (non-reserved devices)

Unmanaged/unknown clients receive DHCP from:

- `10.59.250.201 – 10.59.250.250`

This keeps “random DHCP” near the end of the address space (and away from the octet-grouped ranges).

## Reservation status codes (from router export CSV)

The CSV export uses a simple status indicator per MAC:

- `Y` = reserved + lease present and IPs match (online and correct)
- `N` = reserved + lease present but IP differs (investigate)
- `R` = lease only (no reservation) — candidate to reserve or ignore
- `O` = reservation only (no current lease) — reserved but currently offline
- `X` = name only (custom name exists but no reservation/lease)

# 01 – Network Overview

## Purpose

This repository documents the current home LAN design following migration from `192.168.1.0/24` to `10.59.0.0/16`, and establishes the foundation for segmentation using firewall rules (no VLANs).

The design prioritises:

- Stability over VLAN complexity
- Clean IP grouping by function
- Centralised DHCP control on the router
- DNS resilience
- Future segmentation via allow/deny rules and octet-based logic
- Guest isolation via Guest Network Pro
